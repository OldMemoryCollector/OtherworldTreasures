using System.Text.Json;
using System.Text.Encodings.Web;
using Godot;
using HarmonyLib;
using MegaCrit.Sts2.Core.Modding;
using MegaCrit.Sts2.Core.Models;
using MegaCrit.Sts2.Core.Models.Characters;
using MegaCrit.Sts2.Core.Entities.Players;
using MegaCrit.Sts2.Core.Entities.Cards;
using MegaCrit.Sts2.Core.Runs;
using MegaCrit.Sts2.Core.Unlocks;
using MegaCrit.Sts2.Core.Localization;
using OtherworldTreasures.Scripts.Potions;

[ModInitializer(nameof(Init))]
public static class PotionVisualProbe
{
 public static async void Init()
 {
  var tree=(SceneTree)Engine.GetMainLoop();
  var output=System.Environment.GetEnvironmentVariable("OTHERWORLD_WIKI_OUTPUT") ?? throw new Exception("Set OTHERWORLD_WIKI_OUTPUT");
  Directory.CreateDirectory(Path.Combine(output,"images"));
  var entries=new List<object>();var errors=new List<string>();
  string Text(LocString? s)=>s?.GetFormattedText()??"";
  void Image(Texture2D? texture,string file){if(System.Environment.GetEnvironmentVariable("OTHERWORLD_WIKI_REUSE_IMAGES")=="1"&&File.Exists(Path.Combine(output,"images",file)))return;if(texture==null)throw new Exception("Missing texture");using var image=texture.GetImage();if(image.SavePng(Path.Combine(output,"images",file))!=Error.Ok)throw new Exception("Image save failed");}
  try
  {
   for(int frame=0;frame<180;frame++)await tree.ToSignal(tree,SceneTree.SignalName.ProcessFrame);
   var owner=Player.CreateForNewRun<Ironclad>(UnlockState.all,1UL);var run=RunState.CreateForTest([owner],seed:"WIKICATALOG");
   var assembly=typeof(SCP294Potion).Assembly;
   foreach(var model in ModelDb.All.Where(x=>x.GetType().Assembly==assembly).OrderBy(x=>x.Id.Entry).ToArray())
   {
    var type=model.GetType();var kind=model switch{RelicModel=>"relics",CardModel=>"cards",PotionModel=>"potions",EventModel=>"events",EnchantmentModel=>"enchantments",PowerModel=>"powers",_=>""};if(kind=="")continue;
    var data=new Dictionary<string,object?>{["id"]=model.Id.Entry,["class"]=type.Name,["kind"]=kind};
    try
    {
     string? file=null;
     if(model is RelicModel relic)
     {
      var item=relic.ToMutable();item.Owner=owner;data["title"]=Text(item.Title);data["description"]=Text(item.DynamicDescription);data["flavor"]=relic.Flavor.Exists()?Text(relic.Flavor):"";data["rarity"]=relic.Rarity.ToString();
      file=model.Id.Entry+".png";Image(relic.Icon,file);
     }
     else if(model is CardModel card)
     {
      var item=card.ToMutable();item.Owner=owner;data["title"]=item.Title;data["description"]=item.GetDescriptionForPile(PileType.None);data["rarity"]=item.Rarity.ToString();data["type"]=item.Type.ToString();data["target"]=item.TargetType.ToString();
      data["cost"]=AccessTools.Property(typeof(CardModel),"CanonicalEnergyCost").GetValue(item)?.ToString();data["maxUpgrades"]=item.MaxUpgradeLevel;
      var tiers=new List<object>();for(int i=0;i<Math.Min(item.MaxUpgradeLevel,6);i++){item.UpgradeInternal();tiers.Add(new{level=i+1,title=item.Title,description=item.GetDescriptionForPile(PileType.None),cost=item.EnergyCost.GetResolved()});}data["upgrades"]=tiers;
      file=model.Id.Entry+".png";Image(card.Portrait,file);
     }
     else if(model is PotionModel potion)
     {
      var item=potion.ToMutable();item.Owner=owner;data["title"]=Text(item.Title);data["description"]=Text(item.DynamicDescription);data["rarity"]=item.Rarity.ToString();data["target"]=item.TargetType.ToString();file=model.Id.Entry+".png";Image(potion.Image,file);
     }
     else if(model is EventModel evt)
     {
      data["title"]=Text(evt.Title);
      var profile=AccessTools.Property(type,"AssetProfile")?.GetValue(evt);var portrait=profile?.GetType().GetProperty("InitialPortraitPath")?.GetValue(profile) as string;
      if(!string.IsNullOrEmpty(portrait)){file=model.Id.Entry+".png";Image(ResourceLoader.Load<Texture2D>(portrait),file);}
     }
     else if(model is EnchantmentModel enchantment)
     {
      data["title"]=Text(enchantment.Title);data["description"]=Text(enchantment.DynamicDescription);file=model.Id.Entry+".png";Image(enchantment.Icon,file);
     }
     else if(model is PowerModel power)
     {
      data["title"]=Text(power.Title);var powerInstance=power.ToMutable();AccessTools.PropertySetter(typeof(PowerModel),"Owner").Invoke(powerInstance,[owner.Creature]);var loc=powerInstance.Description;powerInstance.DynamicVars.AddTo(loc);loc.Add("Amount",1);data["description"]=Text(loc);file=model.Id.Entry+".png";Image(power.Icon,file);
     }
     data["image"]=file;
    }
    catch(Exception e){errors.Add(type.Name+": "+e.GetBaseException().Message);data["error"]=e.GetBaseException().Message;}
    entries.Add(data);
   }
   foreach(var relic in ModelDb.AllRelics.Where(x=>x.GetType().Assembly!=assembly).ToArray())
   {
    foreach(var metal in new[]{"golden_relics","silver_relics"})
    {
     try
     {
      var item=relic.ToMutable();item.Owner=owner;
      bool eligible=metal=="golden_relics"?OtherworldTreasures.Scripts.Events.GoldenRelics.CanEnhance(item):OtherworldTreasures.Scripts.Events.SilverRelics.CanEnhance(item);
      if(!eligible)continue;
      var original=Text(item.DynamicDescription);var originalTitle=Text(item.Title);
      bool applied=metal=="golden_relics"?OtherworldTreasures.Scripts.Events.GoldenRelics.Enhance(item):OtherworldTreasures.Scripts.Events.SilverRelics.Enhance(item);
      if(!applied)throw new Exception("Eligible enhancement rejected");
      var file="NATIVE_RELIC_"+relic.Id.Entry+".png";if(!File.Exists(Path.Combine(output,"images",file)))Image(relic.Icon,file);
      entries.Add(new Dictionary<string,object?>{["id"]=metal+"_"+relic.Id.Entry,["baseId"]=relic.Id.Entry,["class"]=relic.GetType().Name,["kind"]=metal,["title"]=Text(item.Title),["description"]=Text(item.DynamicDescription),["baseDescription"]=original,["baseTitle"]=originalTitle,["rarity"]=relic.Rarity.ToString(),["image"]=file});
     }
     catch(Exception e){errors.Add(metal+"/"+relic.Id.Entry+": "+e.GetBaseException().Message);}
    }
   }
   File.WriteAllText(Path.Combine(output,"catalog.json"),JsonSerializer.Serialize(new{entries,errors},new JsonSerializerOptions{WriteIndented=true,Encoder=JavaScriptEncoder.UnsafeRelaxedJsonEscaping}));
   GD.Print("WIKI EXPORT "+entries.Count+" models, "+errors.Count+" errors");tree.Quit(errors.Count==0?0:1);
  }
  catch(Exception e){File.WriteAllText(Path.Combine(output,"export-error.txt"),e.ToString());tree.Quit(1);}
 }
}
